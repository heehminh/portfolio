데이터 품질 검증 방식 보기

이관 후 오류가 발생하는 유형(날짜 역전, NULL 값, 중복키, 금액 이상치)을 SQL 검증으로 자동화했습니다. 

-- 날짜 역전 검증
SELECT 거래ID, 거래일자, 정산일자
FROM 정산정보
WHERE CAST(거래일자 AS DATE) > CAST(정산일자 AS DATE);

-- 누적 금액 이상치 검증
SELECT 거래ID, 누적금액,
  LAG(누적금액) OVER (ORDER BY 거래일자) AS 이전금액
FROM 거래정보
HAVING 누적금액 < 이전금액;
